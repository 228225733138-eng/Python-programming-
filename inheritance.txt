class Person:
    def __init__(self, name):
        self.name = name

    def display(self):
        print("Name:", self.name)

class Student(Person):
    def __init__(self, name, roll_no):
        super().__init__(name)
        self.roll_no = roll_no

    def display_student(self):
        self.display()
        print("Roll No:", self.roll_no)

Student("Ravi", 11).display_student()
