class Vehicle:
    def __init__(self, brand): self.brand = brand
    def start(self): print(self.brand, "is starting")

class Car(Vehicle):
    def __init__(self, brand, doors):
        super().__init__(brand)
        self.doors = doors
    def display(self): print("Brand:", self.brand, "Doors:", self.doors)

car = Car("Tata", 4)
car.start()
car.display()
