class Book:
    def __init__(self, book_id, title, author):
        self.book_id = book_id
        self.title = title
        self.author = author
        self.available = True

    def issue(self):
        if self.available:
            self.available = False
            print("Book issued")
        else:
            print("Book is not available")

    def return_book(self):
        self.available = True
        print("Book returned")

    def display(self):
        print(self.book_id, self.title, self.author, "Available:", self.available)

book = Book(101, "Python Basics", "ABC")
book.display()
book.issue()
book.display()
book.return_book()
