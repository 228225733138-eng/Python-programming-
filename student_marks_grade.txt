class Student:
    def __init__(self, name, roll_no, marks):
        self.name = name
        self.roll_no = roll_no
        self.marks = marks

    def result(self):
        total = sum(self.marks)
        average = total / len(self.marks)
        if average >= 90:
            grade = "A"
        elif average >= 75:
            grade = "B"
        elif average >= 60:
            grade = "C"
        else:
            grade = "D"
        print("Name:", self.name)
        print("Roll No:", self.roll_no)
        print("Total:", total)
        print("Average:", average)
        print("Grade:", grade)

Student("KPRIT", "S11", [91, 93, 88, 98, 90]).result()
